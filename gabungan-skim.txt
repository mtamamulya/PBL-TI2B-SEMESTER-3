<?php

namespace App\Models;

use Illuminate\Database\Eloquent\Model;
use Illuminate\Database\Eloquent\Relations\BelongsTo;
use Illuminate\Database\Eloquent\Relations\HasMany;
use Illuminate\Validation\ValidationException;

class KelompokBidang extends Model
{
    protected $table = 'kelompok_bidang';

    protected $primaryKey = 'id_kelompok_bidang';

    protected $fillable = ['id_rujukan', 'nama_kelompok_bidang'];

    protected static function booted(): void
    {
        static::saving(function (self $bidang) {
            if ($bidang->exists && $bidang->id_rujukan !== null
                && (int) $bidang->id_rujukan === (int) $bidang->getKey()) {
                throw ValidationException::withMessages(['id_rujukan' => 'Kelompok bidang tidak boleh merujuk ke dirinya sendiri.']);
            }
        });
    }

    public function rujukan(): BelongsTo
    {
        return $this->belongsTo(self::class, 'id_rujukan', 'id_kelompok_bidang');
    }

    public function turunan(): HasMany
    {
        return $this->hasMany(self::class, 'id_rujukan', 'id_kelompok_bidang');
    }

    public function kegiatan(): HasMany
    {
        return $this->hasMany(Kegiatan::class, 'id_kelompok_bidang', 'id_kelompok_bidang');
    }
}


===== app\Models\Kegiatan.php =====
<?php

namespace App\Models;

use Illuminate\Database\Eloquent\Builder;
use Illuminate\Database\Eloquent\Model;
use Illuminate\Database\Eloquent\Relations\BelongsTo;
use Illuminate\Database\Eloquent\Relations\HasMany;
use Illuminate\Database\Eloquent\Relations\HasOne;
use Illuminate\Database\Query\Builder as QueryBuilder;
use Illuminate\Support\Carbon;
use Illuminate\Support\Facades\DB;
use Illuminate\Support\Facades\Schema;
use Illuminate\Validation\ValidationException;

class Kegiatan extends Model
{
    protected $table = 'kegiatan';

    protected $primaryKey = 'id_kegiatan';

    protected $fillable = [
        'id_kategori', 'id_user', 'id_skim', 'id_kelompok_bidang',
        'nama_kegiatan', 'dosen_pendamping', 'nomor_sk', 'tanggal_sk',
    ];

    protected function casts(): array
    {
        return [
            'tanggal_sk' => 'date',
        ];
    }

    protected static function booted(): void
    {
        static::updating(function (self $activity) {
            if ($activity->isDirty('id_kategori') && $activity->detail()->exists()) {
                throw ValidationException::withMessages(['id_kategori' => 'Kategori kegiatan yang sudah memiliki jawaban tidak dapat langsung diubah.']);
            }
        });
    }

    public function statusTerkini(): HasOne
    {
        return $this->hasOne(Timeline::class, 'id_kegiatan', 'id_kegiatan')
            ->latestOfMany('tanggal_status');
    }

    public function timeline(): HasMany
    {
        return $this->hasMany(Timeline::class, 'id_kegiatan', 'id_kegiatan')
            ->orderBy('tanggal_status')->orderBy('id_timeline');
    }

    public function scopeUntukUser(Builder $query, ?User $user): Builder
    {
        if (! $user) {
            return $query->whereRaw('1 = 0');
        }

        return $query->where(fn (Builder $q) => $q->where('id_user', $user->getKey())
            ->orWhereHas('anggota', fn (Builder $q) => $q->where('id_user', $user->getKey())));
    }

    public function scopeUntukPrestasi(Builder $query, ?User $pemilik): Builder
    {
        return $query->untukUser($pemilik)
            ->whereHas('statusTerkini', fn (Builder $status) => $status->where('status', 'diterima'));
    }

    /** Satu baris per kegiatan, termasuk kegiatan yang belum memiliki tanggal. */
    public static function tabelDenganTanggal(): QueryBuilder
    {
        // Older installations still retain physical date columns. Exclude them
        // so the derived table exposes each detail-based date exactly once.
        $columns = array_map(
            fn (string $column) => 'kegiatan.'.$column,
            array_values(array_diff(Schema::getColumnListing('kegiatan'), ['tanggal_mulai', 'tanggal_selesai']))
        );
        $kegiatan = DB::table('kegiatan')->select($columns);
        foreach (['tanggal_mulai', 'tanggal_selesai'] as $input) {
            $tanggal = DB::table('detail_kegiatan as d')
                ->join('isian_kategori as i', 'i.id_isian_kategori', '=', 'd.id_isian_kategori')
                ->whereColumn('d.id_kegiatan', 'kegiatan.id_kegiatan')
                ->where('i.input', $input)
                ->orderBy('d.id_detail_kegiatan')->limit(1)->select('d.isian');
            $kegiatan->selectSub($tanggal, $input);
        }

        return $kegiatan;
    }

    public function scopeDenganTanggal(Builder $query): Builder
    {
        return $query->fromSub(self::tabelDenganTanggal(), 'kegiatan');
    }

    private function tanggalDetail(string $input): ?Carbon
    {
        $nilai = array_key_exists($input, $this->attributes)
            ? $this->attributes[$input]
            : $this->nilaiIsian($input);

        return $nilai && Carbon::hasFormat($nilai, 'Y-m-d') ? Carbon::createFromFormat('!Y-m-d', $nilai) : null;
    }

    public function getTanggalMulaiAttribute(): ?Carbon
    {
        return $this->tanggalDetail('tanggal_mulai');
    }

    public function getTanggalSelesaiAttribute(): ?Carbon
    {
        return $this->tanggalDetail('tanggal_selesai');
    }

    public function kategori(): BelongsTo
    {
        return $this->belongsTo(Kategori::class, 'id_kategori');
    }

    public function skim(): BelongsTo
    {
        return $this->belongsTo(Skim::class, 'id_skim', 'id_skim');
    }

    public function kelompokBidang(): BelongsTo
    {
        return $this->belongsTo(KelompokBidang::class, 'id_kelompok_bidang', 'id_kelompok_bidang');
    }

    public function detail(): HasMany
    {
        return $this->hasMany(DetailKegiatan::class, 'id_kegiatan', 'id_kegiatan')->orderBy('id_detail_kegiatan');
    }

    public function prestasi(): HasMany
    {
        return $this->hasMany(Prestasi::class, 'id_kegiatan', 'id_kegiatan');
    }

    public function user(): BelongsTo
    {
        return $this->belongsTo(User::class, 'id_user', 'id_user');
    }

    public function berita(): HasMany
    {
        return $this->hasMany(Berita::class, 'id_kegiatan', 'id_kegiatan');
    }

    public function nilaiIsian(string $input): ?string
    {
        $details = $this->relationLoaded('detail') ? $this->detail : $this->detail()->with('isianKategori')->get();

        return $details->first(fn ($detail) => $detail->isianKategori?->input === $input)?->isian;
    }

    public function dokumen(): HasMany
    {
        return $this->hasMany(Dokumen::class, 'id_kegiatan')->orderBy('id_dokumen');
    }

    public function anggota(): HasMany
    {
        return $this->hasMany(AnggotaKegiatan::class, 'id_kegiatan');
    }
}

<?php

namespace App\BackendKel1\JenisDokumen;

use App\Http\Controllers\Controller;
use App\Models\JenisDokumen;
use App\Support\ApiResponse;
use Illuminate\Http\JsonResponse;
use Illuminate\Http\Request;
use Illuminate\Validation\Rule;

/** CRUD /api/v1/jenis-dokumen */
class JenisDokumenController extends Controller
{
    public function index(Request $request): JsonResponse
    {
        $request->validate([
            'per_page' => ['nullable', 'integer', 'min:1', 'max:100'],
            'q' => ['nullable', 'string', 'max:100'],
        ]);

        $p = JenisDokumen::query()
            ->when($request->filled('q'), fn ($q) => $q->where('nama_jenis_dokumen', 'like', '%'.$request->query('q').'%'))
            ->orderByDesc('id_jenis_dokumen')
            ->paginate((int) $request->query('per_page', 15));

        return ApiResponse::paginated($p, $p->getCollection()->map(fn ($item) => self::bentuk($item))->all());
    }

    public function show(int $id): JsonResponse
    {
        $item = JenisDokumen::query()->find($id);

        return $item ? ApiResponse::ok(self::bentuk($item)) : ApiResponse::gagal('Jenis dokumen tidak ditemukan');
    }

    public function store(Request $request): JsonResponse
    {
        $data = $request->validate([
            'nama_jenis_dokumen' => ['required', 'string', 'max:100', 'unique:jenis_dokumen,nama_jenis_dokumen'],
            'keterangan' => ['nullable', 'string', 'max:500'],
        ]);

        $item = JenisDokumen::query()->create($data);

        return ApiResponse::ok(self::bentuk($item));
    }

    public function update(Request $request, int $id): JsonResponse
    {
        $item = JenisDokumen::query()->find($id);

        if (! $item) {
            return ApiResponse::gagal('Jenis dokumen tidak ditemukan');
        }

        $data = $request->validate([
            'nama_jenis_dokumen' => [
                'required',
                'string',
                'max:100',
                Rule::unique('jenis_dokumen', 'nama_jenis_dokumen')->ignore($item->id_jenis_dokumen, 'id_jenis_dokumen'),
            ],
            'keterangan' => ['nullable', 'string', 'max:500'],
        ]);

        $item->update($data);

        return ApiResponse::ok(self::bentuk($item));
    }

    public function destroy(int $id): JsonResponse
    {
        $item = JenisDokumen::query()->find($id);

        if (! $item) {
            return ApiResponse::gagal('Jenis dokumen tidak ditemukan');
        }

        if ($item->dokumen()->exists() || $item->buktiPrestasi()->exists()) {
            return ApiResponse::gagal('Jenis dokumen masih digunakan dan tidak dapat dihapus');
        }

        $item->delete();

        return ApiResponse::ok([]);
    }

    public static function bentuk(JenisDokumen $item): array
    {
        return [
            'id_jenis_dokumen' => $item->id_jenis_dokumen,
            'nama_jenis_dokumen' => $item->nama_jenis_dokumen,
            'keterangan' => $item->keterangan,
        ];
    }
}

<?php

namespace App\Http\Controllers;

use App\Support\ApiClient;
use App\Support\ApiException;
use Illuminate\Http\JsonResponse;
use Illuminate\Http\Request;

class JenisDokumenController extends Controller
{
    public function index()
    {
        $hasil = ApiClient::pengguna()->get('/jenis-dokumen', ['per_page' => 100]);
        $rows = collect($hasil['data'] ?? [])->map(fn (array $item) => $this->bentukBaris($item))->all();

        return view('admin.jenis-dokumen', ['rows' => $rows]);
    }

    public function store(Request $request): JsonResponse
    {
        $data = $this->validasi($request);

        try {
            $hasil = ApiClient::pengguna()->post('/jenis-dokumen', [
                'nama_jenis_dokumen' => $data['name'],
                'keterangan' => $data['description'] ?? null,
            ]);
        } catch (ApiException $e) {
            return $this->responGagalApi($e);
        }

        return response()->json([
            'ok' => true,
            'row' => $this->bentukBaris($hasil['data'] ?? []),
        ]);
    }

    public function update(Request $request, string $id): JsonResponse
    {
        $data = $this->validasi($request);

        try {
            $hasil = ApiClient::pengguna()->put("/jenis-dokumen/{$id}", [
                'nama_jenis_dokumen' => $data['name'],
                'keterangan' => $data['description'] ?? null,
            ]);
        } catch (ApiException $e) {
            return $this->responGagalApi($e);
        }

        return response()->json([
            'ok' => true,
            'row' => $this->bentukBaris($hasil['data'] ?? []),
        ]);
    }

    public function destroy(string $id): JsonResponse
    {
        try {
            ApiClient::pengguna()->delete("/jenis-dokumen/{$id}");
        } catch (ApiException $e) {
            return $this->responGagalApi($e);
        }

        return response()->json(['ok' => true]);
    }

    private function validasi(Request $request): array
    {
        return $request->validate([
            'name' => ['required', 'string', 'max:100'],
            'description' => ['nullable', 'string', 'max:255'],
        ], [
            'name.required' => 'Nama jenis dokumen wajib diisi.',
            'name.max' => 'Nama jenis dokumen maksimal 100 karakter.',
        ]);
    }

    /** @param array<string, mixed> $item */
    private function bentukBaris(array $item): array
    {
        return [
            'id' => $item['id_jenis_dokumen'] ?? null,
            'name' => $item['nama_jenis_dokumen'] ?? '',
            'description' => $item['keterangan'] ?? null,
        ];
    }

    private function responGagalApi(ApiException $e): JsonResponse
    {
        $status = method_exists($e, 'getStatusCode')
            ? $e->getStatusCode()
            : ($e->status ?? ($e->getCode() ?: 422));

        if ($status < 400 || $status > 599) {
            $status = 422;
        }

        return response()->json([
            'message' => $e->getMessage(),
        ], $status);
    }
}

{{-- Hapus baris jQuery ini kalau layout admin kamu sudah memuat jQuery 3.x --}}
<script src="https://code.jquery.com/jquery-3.7.1.min.js"></script>
<script src="https://cdn.datatables.net/2.1.8/js/dataTables.min.js"></script>
<script>
(() => {
    const URL_STORE  = @json(route('admin.kegiatan.jenis-dokumen.store'));
    const URL_UPDATE = @json(route('admin.kegiatan.jenis-dokumen.update', ['id' => '__ID__']));
    const URL_DELETE = @json(route('admin.kegiatan.jenis-dokumen.destroy', ['id' => '__ID__']));
    const CSRF = @json(csrf_token());

    const rows = @json($rows);
    let editingId = null; // null = tambah baru
    let idHapus = null;

    const $ = (id) => document.getElementById(id);
    const esc = (s) => String(s ?? '').replace(/[&<>"']/g, (c) =>
        ({ '&': '&amp;', '<': '&lt;', '>': '&gt;', '"': '&quot;', "'": '&#39;' }[c]));

    const IKON_EDIT = '<svg width="20" height="20" viewBox="0 0 24 24" fill="none" stroke="currentColor" stroke-width="1.5" stroke-linecap="round" stroke-linejoin="round"><path d="M4 20h4L18.5 9.5a2.828 2.828 0 1 0-4-4L4 16v4M13.5 6.5l4 4"/></svg>';
    const IKON_HAPUS = '<svg width="20" height="20" viewBox="0 0 24 24" fill="none" stroke="currentColor" stroke-width="1.5" stroke-linecap="round" stroke-linejoin="round"><path d="M4 7l16 0" /><path d="M10 11l0 6" /><path d="M14 11l0 6" /><path d="M5 7l1 12a2 2 0 0 0 2 2h8a2 2 0 0 0 2 -2l1 -12" /><path d="M9 7v-3a1 1 0 0 1 1 -1h6a1 1 0 0 1 1 1v3" /></svg>';

    // ---------- DataTables ----------
    const table = new DataTable('#tabel', {
        data: rows,
        autoWidth: false,
        pageLength: 10,
        order: [],                       // tanpa urutan awal = urutan asli dari server
        columns: [
            { data: null, orderable: false, searchable: false, defaultContent: '' },   // No: diisi otomatis saat draw
            { data: 'name',        className: 'truncate', createdCell: (td, d) => { td.title = d ?? ''; },
              render: (d, type) => type === 'display' ? esc(d) : (d ?? '') },
            { data: 'description', className: 'truncate', createdCell: (td, d) => { td.title = d ?? ''; },
              render: (d, type) => type === 'display' ? esc(d) : (d ?? '') },
            {
                data: null, orderable: false, searchable: false,
                render: (d, type, r) => `
                    <div class="flex items-center gap-3">
                        <button type="button" data-act="edit" data-id="${esc(r.id)}" title="Edit" aria-label="Edit"
                                class="rounded-md bg-[#CDDEF9] p-1.5 text-[#2277FF] hover:bg-[#BBD2F6] cursor-pointer">${IKON_EDIT}</button>
                        <button type="button" data-act="hapus" data-id="${esc(r.id)}" title="Hapus" aria-label="Hapus"
                                class="rounded-md bg-[#FFD9D9] p-1.5 text-[#E5484D] hover:bg-[#FFC9C9] cursor-pointer">${IKON_HAPUS}</button>
                    </div>`
            }
        ],
        // Hanya paginasi (tengah bawah). Kotak cari, "show entries", dan info bawaan dimatikan
        layout: { topStart: null, topEnd: null, bottomStart: null, bottomEnd: null, bottom: 'paging' },
        pagingType: 'simple_numbers',
        language: {
            emptyTable: 'Tidak ada jenis dokumen.',
            zeroRecords: 'Tidak ada jenis dokumen.',
            paginate: { previous: 'Sebelumnya', next: 'Berikutnya' }
        }
    });

    // Nomor urut mengikuti tampilan (urutan + halaman saat ini)
    function isiNomor() {
        const mulai = table.page.info().start;
        table.column(0, { page: 'current' }).nodes().each((cell, i) => { cell.textContent = mulai + i + 1; });
    }
    // Sembunyikan paginasi kalau hanya ada 1 halaman
    function togglePager() {
        const p = document.querySelector('#tabel-wrap .dt-paging');
        if (p) p.style.display = table.page.info().pages > 1 ? '' : 'none';
    }
    table.on('draw.dt', () => { isiNomor(); togglePager(); });
    isiNomor();
    togglePager();

    const cariBaris = (id) => table.row((idx, data) => String(data.id) === String(id));

    // Setelah hapus: kalau halaman yang sedang dibuka jadi kosong, mundur ke halaman terakhir yang masih ada isinya
    function pindahJikaHalamanKosong() {
        const i = table.page.info();
        if (i.pages > 0 && i.page >= i.pages) table.page(i.pages - 1).draw('page');
    }

    // ---------- Pop-up tambah / edit ----------
    function bukaModal(row) {
        editingId = row ? row.id : null;
        $('jd-modal-judul').textContent = row ? 'Edit Jenis Dokumen' : 'Tambah Jenis Dokumen';
        $('jd-modal-sub').textContent = row
            ? 'Ubah data jenis dokumen program studi'
            : 'Isi data jenis dokumen baru program studi';
        $('jd-nama').value = row ? row.name : '';
        $('jd-ket').value = row ? row.description : '';
        tampilError('');
        $('jd-modal').classList.remove('hidden');
        $('jd-modal').classList.add('flex');
        $('jd-nama').focus();
    }

    function tutupModal() {
        $('jd-modal').classList.add('hidden');
        $('jd-modal').classList.remove('flex');
    }

    function tampilError(pesan) {
        const el = $('jd-error');
        el.textContent = pesan;
        el.classList.toggle('hidden', !pesan);
    }

    async function kirim(url, method, body) {
        const res = await fetch(url, {
            method,
            headers: {
                'X-CSRF-TOKEN': CSRF,
                'Accept': 'application/json',
                'Content-Type': 'application/json',
            },
            body: body ? JSON.stringify(body) : undefined,
        });
        const json = await res.json().catch(() => ({}));
        if (!res.ok) {
            const pesan = json.errors ? Object.values(json.errors)[0][0] : (json.message || 'Terjadi kesalahan.');
            throw new Error(pesan);
        }
        return json;
    }

    async function simpan() {
        const name = $('jd-nama').value.trim();
        const description = $('jd-ket').value.trim();
        if (!name) { tampilError('Nama jenis dokumen wajib diisi.'); return; }

        const tombol = $('jd-simpan');
        tombol.disabled = true;
        try {
            if (editingId === null) {
                const { row } = await kirim(URL_STORE, 'POST', { name, description });
                table.order([]);                       // baris baru muncul di urutan akhir
                table.row.add(row).draw();
                table.page('last').draw('page');       // lompat ke halaman terakhir
            } else {
                const { row } = await kirim(URL_UPDATE.replace('__ID__', editingId), 'PUT', { name, description });
                cariBaris(row.id).data(row).draw(false);   // tetap di halaman yang sama
            }
            tutupModal();
        } catch (e) {
            tampilError(e.message);
        } finally {
            tombol.disabled = false;
        }
    }

    // ---------- Modal konfirmasi hapus ----------
    const modalHapus = $('modalHapus'), btnHapusYa = $('modalHapusYa');

    function bukaHapus(id) {
        const r = cariBaris(id).data();
        if (!r) return;
        idHapus = id;
        if (typeof modalHapus.showModal !== 'function') {          // cadangan untuk browser lama
            if (confirm(`Hapus "${r.name}"?`)) prosesHapus();
            return;
        }
        $('modalHapusNama').textContent = `"${r.name}"`;
        $('modalHapusError').hidden = true;
        modalHapus.showModal();                                    // fokus awal di tombol Batal; Esc menutup modal
    }

    async function prosesHapus() {
        if (idHapus === null) return;
        btnHapusYa.disabled = true;                                // cegah klik ganda
        btnHapusYa.textContent = 'Menghapus...';
        try {
            await kirim(URL_DELETE.replace('__ID__', idHapus), 'DELETE');
            cariBaris(idHapus).remove().draw(false);
            pindahJikaHalamanKosong();                                 // hapus data terakhir di halaman 2 -> otomatis ke halaman 1
            if (modalHapus.open) modalHapus.close();
        } catch (e) {
            $('modalHapusError').textContent = e.message;
            $('modalHapusError').hidden = false;
            btnHapusYa.disabled = false;
            btnHapusYa.textContent = 'Hapus';
        }
    }

    $('modalHapusBatal').addEventListener('click', () => modalHapus.close());
    modalHapus.addEventListener('click', (e) => { if (e.target === modalHapus) modalHapus.close(); });   // klik area gelap = batal
    modalHapus.addEventListener('close', () => { idHapus = null; btnHapusYa.disabled = false; btnHapusYa.textContent = 'Hapus'; });
    btnHapusYa.addEventListener('click', prosesHapus);

    // ---------- Event ----------
    $('jd-tambah').addEventListener('click', () => bukaModal(null));
    $('jd-batal').addEventListener('click', tutupModal);
    $('jd-simpan').addEventListener('click', simpan);
    $('jd-modal').addEventListener('click', (e) => { if (e.target === $('jd-modal')) tutupModal(); });
    document.addEventListener('keydown', (e) => { if (e.key === 'Escape') tutupModal(); });
    // Enter menyimpan hanya dari kolom nama (kolom keterangan adalah textarea: Enter = baris baru)
    $('jd-nama').addEventListener('keydown', (e) => { if (e.key === 'Enter') simpan(); });

    // Cari di kolom Nama Jenis Dokumen
    $('jd-cari').addEventListener('input', (e) => table.column(1).search(e.target.value.trim()).draw());

    $('tabel').addEventListener('click', (e) => {
        const b = e.target.closest('button[data-act]');
        if (!b) return;
        if (b.dataset.act === 'edit') { const r = cariBaris(b.dataset.id).data(); if (r) bukaModal(r); }

(function () {
    var rows = @json($rows);

    var $ = function (id) { return document.getElementById(id); };
    var q = '';

    var TXT = DataTable.render.text(); // otomatis escape HTML

    /* ---------- Pencarian: nama jenis dokumen ---------- */
    DataTable.ext.search.push(function (settings, searchData, dataIndex, r) {
        if (settings.nTable.id !== 'tabel') return true;
        return String(r.name == null ? '' : r.name).toLowerCase().indexOf(q) !== -1;
    });

    var table = new DataTable('#tabel', {
        data: rows,
        autoWidth: false,
        pageLength: 10,
        order: [],                       // tanpa urutan awal = urutan asli dari server
        columns: [
            { data: null, orderable: false, searchable: false, defaultContent: '' },   // nomor urut: diisi ulang setiap tabel digambar
            { data: 'name', render: TXT, className: 'truncate font-semibold' },
            { data: 'description', render: TXT, className: 'truncate' }
        ],
        // Tooltip: arahkan kursor ke teks yang terpotong "..." -> muncul teks lengkapnya
        createdRow: function (tr, r) {
            var td = tr.children;
            td[1].title = r.name || '';
            td[2].title = r.description || '';
        },
        // Hanya paginasi (tengah bawah). Kotak cari, "show entries", dan info bawaan dimatikan
        layout: { topStart: null, topEnd: null, bottomStart: null, bottomEnd: null, bottom: 'paging' },
        pagingType: 'simple_numbers',
        language: {
            emptyTable: 'Tidak ada jenis dokumen.',
            zeroRecords: 'Tidak ada jenis dokumen.',
            paginate: { previous: 'Sebelumnya', next: 'Berikutnya' }
        }
    });

    // Nomor urut mengikuti hasil pencarian & halaman (halaman 2 mulai dari 11, dst.)
    function isiNomor() {
        var mulai = table.page.info().start;
        table.column(0, { page: 'current' }).nodes().each(function (cell, i) { cell.textContent = mulai + i + 1; });
    }

    // Sembunyikan paginasi kalau hanya ada 1 halaman
    function togglePager() {
        var p = document.querySelector('#tabel-wrap .dt-paging');
        if (p) p.style.display = table.page.info().pages > 1 ? '' : 'none';
    }
    table.on('draw.dt', function () { isiNomor(); togglePager(); });
    isiNomor();
    togglePager();

    /* ---------- Event pencarian ---------- */
    $('jd-cari').addEventListener('input', function (e) { q = e.target.value.trim().toLowerCase(); table.draw(); });
})();

<?php

namespace Tests\Feature;

use App\Models\JenisDokumen;
use App\Models\User;
use Database\Seeders\JenisDokumenSeeder;
use Illuminate\Foundation\Testing\DatabaseMigrations;
use Illuminate\Support\Facades\DB;
use Illuminate\Support\Facades\Http;
use Tests\TestCase;

class KaprodiJenisDokumenTest extends TestCase
{
    use DatabaseMigrations;

    private function palsukanApiJenisDokumen(): void
    {
        Http::fake([
            '*/jenis-dokumen*' => function () {
                $data = JenisDokumen::query()
                    ->get()
                    ->map(fn (JenisDokumen $item) => [
                        'id_jenis_dokumen' => $item->id_jenis_dokumen,
                        'nama_jenis_dokumen' => $item->nama_jenis_dokumen,
                        'keterangan' => $item->keterangan,
                    ])
                    ->all();

                return Http::response([
                    'success' => true,
                    'message' => 'Data berhasil diambil',
                    'data' => $data,
                ], 200);
            },
        ]);
    }

    private function masukSebagaiKaprodi(): User
    {
        DB::table('roles')->insertOrIgnore(['id_role' => 2, 'nama_role' => 'kaprodi']);

        return User::firstOrCreate(
            ['email' => 'kaprodi@trk.ac.id'],
            ['id_role' => 2, 'password' => 'password', 'status' => true]
        );
    }

    public function test_guest_is_redirected_to_login(): void
    {
        $response = $this->get(route('kaprodi.kegiatan.jenis-dokumen'));

        $response->assertRedirect(route('login'));
    }

    public function test_kaprodi_can_view_seeded_document_types_from_database(): void
    {
        $this->seed(JenisDokumenSeeder::class);
        $this->palsukanApiJenisDokumen();

        $response = $this->actingAs($this->masukSebagaiKaprodi())
            ->withSession(['kaprodi' => ['nama' => 'Kaprodi']])
            ->get(route('kaprodi.kegiatan.jenis-dokumen'));

        $response->assertOk();
        $response->assertViewIs('kaprodi.jenis-dokumen');
        $response->assertViewHas('rows');

        $rows = $response->viewData('rows');
        $this->assertCount(5, $rows);
        $this->assertSame('Sertifikat', $rows[0]['name']);
        $this->assertSame('Sertifikat keikutsertaan atau penghargaan kegiatan', $rows[0]['description']);
    }

    public function test_kaprodi_view_reflects_custom_database_document_type(): void
    {
        JenisDokumen::create([
            'nama_jenis_dokumen' => 'Surat Rekomendasi Lomba',
            'keterangan' => 'Surat rekomendasi resmi dari pimpinan jurusan',
        ]);
        $this->palsukanApiJenisDokumen();

        $response = $this->actingAs($this->masukSebagaiKaprodi())
            ->withSession(['kaprodi' => ['nama' => 'Kaprodi']])
            ->get(route('kaprodi.kegiatan.jenis-dokumen'));

        $response->assertOk();
        $rows = $response->viewData('rows');

        $this->assertCount(1, $rows);
        $this->assertSame('Surat Rekomendasi Lomba', $rows[0]['name']);
        $this->assertSame('Surat rekomendasi resmi dari pimpinan jurusan', $rows[0]['description']);
    }
}

<?php

namespace App\Http\Controllers;

use App\Repositories\KurikulumStore;
use App\Repositories\MataKuliahStore;
use Illuminate\Http\Request;

class AdminMataKuliahController extends Controller
{
    public function __construct(private MataKuliahStore $mk, private KurikulumStore $kurikulum) {}

    /** Daftar mata kuliah per kurikulum & semester (?kurikulum=..&semester=..&q=..). */
    public function index(Request $request)
    {
        $daftar = $this->kurikulum->options();
        $ids = array_column($daftar, 'id');

        $kurikulumId = in_array($request->query('kurikulum'), $ids, true) ? $request->query('kurikulum') : ($ids[0] ?? '');

        $semester = (int) $request->query('semester');
        if ($semester < 1 || $semester > 8) {
            $semester = $this->mk->firstSemester($kurikulumId);
        }

        $q = trim((string) $request->query('q'));

        return view('admin.mata-kuliah', [
            'kurikulumList' => $daftar,
            'kurikulumId' => $kurikulumId,
            'semesterList' => range(1, 8),
            'semester' => $semester,
            'q' => $q,
        ] + $this->mk->listing($kurikulumId, $semester, $q));
    }

    /** Form tambah mata kuliah. */
    public function create()
    {
        return view('admin.tambah-mata-kuliah', [
            'kurikulumOptions' => collect($this->kurikulum->options())->pluck('nama', 'id')->all(),
        ]);
    }

    /** Simpan mata kuliah baru. */
    public function store(Request $request)
    {
        $ids = array_column($this->kurikulum->options(), 'id');

        $data = $request->validate([
            'kurikulum' => ['required', 'in:'.implode(',', $ids)],
            'semester' => ['required', 'integer', 'between:1,8'],
            'kode' => ['required', 'string', 'max:30', function ($attribute, $value, $fail) use ($request) {
                if ($this->mk->kodeExists((string) $request->input('kurikulum'), trim($value))) {
                    $fail('Kode mata kuliah sudah dipakai pada kurikulum ini.');
                }
            }],
            'nama' => ['required', 'string', 'max:150'],
            'sks_teori' => ['required', 'integer', 'between:0,99'],
            'sks_praktek' => ['required', 'integer', 'between:0,99'],
            'jam_teori' => ['required', 'integer', 'between:0,99'],
            'jam_praktek' => ['required', 'integer', 'between:0,99'],
        ], [
            'required' => 'Kolom ini wajib diisi.',
            'integer' => 'Harus berupa angka bulat.',
            'between' => 'Nilai harus antara :min sampai :max.',
            'max' => 'Terlalu panjang (maksimal :max karakter).',
            'in' => 'Pilihan tidak valid.',
        ]);

        $this->mk->create($data);

        return redirect()
            ->route('admin.kurikulum.mata-kuliah', ['kurikulum' => $data['kurikulum'], 'semester' => $data['semester']])
            ->with('success', 'Mata kuliah "'.$data['nama'].'" berhasil ditambahkan.');
    }
}

<?php

namespace App\Repositories;

use App\Models\Kurikulum;
use App\Models\Prodi;
use App\Support\KolomOpsional;
use App\Support\LinkFile;
use Illuminate\Http\UploadedFile;

/**
 * Versi DATABASE dari KurikulumStore (tabel `kurikulum`).
 * Nama method dan bentuk array SAMA dengan versi JSON, jadi controller dan view tidak berubah.
 * Dipakai bila DATA_SOURCE=database (lihat AppServiceProvider).
 *
 * Pemetaan: id <- id_kurikulum, name <- nama_kurikulum, description <- deskripsi*, year <- tahun_terbit*,
 *           status (aktif|nonaktif) <- status (boolean), file <- link.
 * (*) kolom usulan; selama belum ada di database dilewati otomatis (KolomOpsional) dan nilainya kosong.
 */
class KurikulumDbStore extends KurikulumStore
{
    public function all(): array
    {
        return Kurikulum::query()->orderByDesc('id_kurikulum')->get()
            ->map(fn (Kurikulum $k) => $this->baris($k))->all();
    }

    public function find(string $id): ?array
    {
        $k = ctype_digit($id) ? Kurikulum::query()->find((int) $id) : null;

        return $k ? $this->baris($k) : null;
    }

    public function create(array $data, ?UploadedFile $pdf = null): array
    {
        $k = new Kurikulum;
        $k->forceFill(KolomOpsional::saring('kurikulum', [
            'id_prodi' => $this->idProdi(),
            'nama_kurikulum' => $data['name'],
            'deskripsi' => ($data['description'] ?? '') !== '' ? $data['description'] : null,
            'tahun_terbit' => ($data['year'] ?? '') !== '' ? (int) $data['year'] : null,
            'status' => true,
            'link' => $pdf ? $pdf->store('kurikulum', 'public') : null,
        ]))->save();

        return $this->baris($k);
    }

    public function update(string $id, array $data, ?UploadedFile $pdf = null): ?array
    {
        $k = ctype_digit($id) ? Kurikulum::query()->find((int) $id) : null;
        if (! $k) {
            return null;
        }

        $k->forceFill(KolomOpsional::saring('kurikulum', [
            'nama_kurikulum' => $data['name'],
            'deskripsi' => ($data['description'] ?? '') !== '' ? $data['description'] : null,
            'tahun_terbit' => ($data['year'] ?? '') !== '' ? (int) $data['year'] : null,
        ]));
        if ($pdf) {
            LinkFile::hapus($k->link);
            $k->link = $pdf->store('kurikulum', 'public');
        }
        $k->save();

        return $this->baris($k);
    }

    public function toggleStatus(string $id): ?array
    {
        $k = ctype_digit($id) ? Kurikulum::query()->find((int) $id) : null;
        if (! $k) {
            return null;
        }

        $k->status = ! $k->status;
        $k->save();

        return $this->baris($k);
    }

    /** Mata kuliah ikut terhapus oleh database (cascadeOnDelete pada mata_kuliah.id_kurikulum). */
    public function delete(string $id): bool
    {
        $k = ctype_digit($id) ? Kurikulum::query()->find((int) $id) : null;
        if (! $k) {
            return false;
        }

        LinkFile::hapus($k->link);
        $k->delete();

        return true;
    }

    public function pdfUrl(array $row): ?string
    {
        return LinkFile::url($row['file'] ?? null);
    }

    private function baris(Kurikulum $k): array
    {
        return [
            'id' => (string) $k->id_kurikulum,
            'name' => $k->nama_kurikulum,
            'description' => (string) $k->getAttribute('deskripsi'),
            'year' => (string) $k->getAttribute('tahun_terbit'),
            'status' => $k->status ? 'aktif' : 'nonaktif',
            'file' => $k->link,
            'created_at' => $k->created_at?->toDateTimeString(),
        ];
    }

    /**
     * kurikulum.id_prodi wajib diisi. Aplikasi ini hanya untuk satu prodi (TRK), jadi dipakai baris prodi pertama.
     * Bila tabel prodi masih kosong, satu baris TRK dibuat otomatis dari config('admin.prodi').
     */
    private function idProdi(): int
    {
        $id = Prodi::query()->orderBy('id_prodi')->value('id_prodi');
        if ($id) {
            return (int) $id;
        }

        $p = config('admin.prodi', []);

        return (int) Prodi::query()->create([
            'kode_prodi' => $p['kode'] ?? 'TRK',
            'nama_prodi' => $p['nama'] ?? 'Teknologi Rekayasa Komputer',
            'jenjang' => $p['jenjang'] ?? 'D4',
        ])->getKey();
    }
}

<?php

namespace App\Repositories;

use Illuminate\Http\UploadedFile;
use Illuminate\Support\Facades\Storage;
use Illuminate\Support\Str;

/**
 * Penyimpanan kurikulum SIMULASI berbasis file JSON (storage/app/private/admin/kurikulum.json).
 * Dipakai dashboard admin (CRUD) dan halaman publik (lewat JsonKurikulumRepository).
 *
 * Saat database siap: ganti isi method di class ini dengan Eloquent (nama method & bentuk
 * array tetap), halaman admin dan publik tidak perlu diubah.
 *
 * Satu baris: id, name, description, year, status (aktif|nonaktif), file (path PDF | null), created_at
 *   - file berawalan "dokumen/" = PDF di folder public/ (data awal)
 *   - selain itu             = PDF hasil upload di disk "public" (storage/app/public)
 */
class KurikulumStore
{
    private const PATH = 'admin/kurikulum.json';

    /** @return array<int, array<string, mixed>> */
    public function all(): array
    {
        $disk = Storage::disk('local');

        if (! $disk->exists(self::PATH)) {
            $this->save($this->seed());
        }

        return json_decode($disk->get(self::PATH), true) ?: [];
    }

    /** Daftar ringkas untuk dropdown (semua status), tahun terbaru dulu. @return array<int, array{id: string, nama: string}> */
    public function options(): array
    {
        $rows = $this->all();
        usort($rows, fn ($a, $b) => strcmp((string) $b['year'], (string) $a['year']) ?: strcmp($a['name'], $b['name']));

        return array_map(fn ($r) => ['id' => (string) $r['id'], 'nama' => $r['name']], $rows);
    }

    public function find(string $id): ?array
    {
        foreach ($this->all() as $row) {
            if ($row['id'] === $id) {
                return $row;
            }
        }

        return null;
    }

    public function create(array $data, ?UploadedFile $pdf = null): array
    {
        $rows = $this->all();

        $row = [
            'id' => $this->uniqueId($data['name'], $rows),
            'name' => $data['name'],
            'description' => $data['description'] ?? '',
            'year' => $data['year'],
            'status' => 'aktif',
            'file' => $pdf ? $pdf->store('kurikulum', 'public') : null,
            'created_at' => now()->toDateTimeString(),
        ];

        array_unshift($rows, $row);
        $this->save($rows);

        return $row;
    }

    public function update(string $id, array $data, ?UploadedFile $pdf = null): ?array
    {
        $rows = $this->all();
        $found = null;

        foreach ($rows as &$row) {
            if ($row['id'] !== $id) {
                continue;
            }
            $row['name'] = $data['name'];
            $row['description'] = $data['description'] ?? '';
            $row['year'] = $data['year'];
            if ($pdf) {
                $this->deleteFile($row['file']);
                $row['file'] = $pdf->store('kurikulum', 'public');
            }
            $found = $row;
        }
        unset($row);

        if ($found) {
            $this->save($rows);
        }

        return $found;
    }

    public function toggleStatus(string $id): ?array
    {
        $rows = $this->all();
        $found = null;

        foreach ($rows as &$row) {
            if ($row['id'] === $id) {
                $row['status'] = $row['status'] === 'aktif' ? 'nonaktif' : 'aktif';
                $found = $row;
            }
        }
        unset($row);

        if ($found) {
            $this->save($rows);
        }

        return $found;
    }

    public function delete(string $id): bool
    {
        $rows = $this->all();
        $keep = [];

        foreach ($rows as $row) {
            if ($row['id'] === $id) {
                $this->deleteFile($row['file']);
                // Mata kuliah milik kurikulum ini ikut terhapus (di database = cascadeOnDelete).
                app(MataKuliahStore::class)->deleteByKurikulum($id);
            } else {
                $keep[] = $row;
            }
        }

        $this->save($keep);

        return count($keep) < count($rows);
    }

    /** URL PDF untuk ditampilkan (atau null bila belum diupload). */
    public function pdfUrl(array $row): ?string
    {
        if (empty($row['file'])) {
            return null;
        }

        return str_starts_with($row['file'], 'dokumen/')
            ? asset($row['file'])
            : Storage::disk('public')->url($row['file']);
    }

    /** Bentuk yang dikirim ke halaman admin. */
    public function forAdmin(array $row): array
    {
        return [
            'id' => $row['id'],
            'name' => $row['name'],
            'description' => $row['description'],
            'year' => $row['year'],
            'status' => $row['status'],
            'pdf_url' => $this->pdfUrl($row),
        ];
    }

    private function save(array $rows): void
    {
        Storage::disk('local')->put(self::PATH, json_encode(array_values($rows), JSON_PRETTY_PRINT | JSON_UNESCAPED_UNICODE | JSON_UNESCAPED_SLASHES));
    }

    private function deleteFile(?string $file): void
    {
        if ($file && ! str_starts_with($file, 'dokumen/')) {
            Storage::disk('public')->delete($file);
        }
    }

    private function uniqueId(string $name, array $rows): string
    {
        $base = Str::slug($name) ?: 'kurikulum';
        $ids = array_column($rows, 'id');
        $id = $base;
        for ($i = 2; in_array($id, $ids, true); $i++) {
            $id = $base.'-'.$i;
        }

        return $id;
    }

    /** Data awal: diambil dari config/kurikulum.php (PDF contoh di public/dokumen/kurikulum/). */
    private function seed(): array
    {
        return array_map(fn ($k) => [
            'id' => $k['id'],
            'name' => $k['nama'],
            'description' => '',
            'year' => preg_match('/\d{4}/', $k['nama'], $m) ? $m[0] : '',
            'status' => 'aktif',
            'file' => $k['file'],
            'created_at' => now()->toDateTimeString(),
        ], config('kurikulum.daftar', []));
    }
}

<?php

namespace App\Repositories;

use App\Models\MataKuliah;

/**
 * Versi DATABASE dari MataKuliahStore (tabel `mata_kuliah`). Nama method dan bentuk array sama dengan versi JSON.
 *
 * Pemetaan: id <- id_mk, kurikulum <- id_kurikulum (string), kode <- kode_mk, nama <- nama_mk,
 *           sks_praktek <- sks_praktik (perhatikan ejaan di database), jam_praktek <- jam_praktek.
 * Method turunan (semesters, bySemester, listing, firstSemester, ...) bekerja dari forKurikulum().
 */
class MataKuliahDbStore extends MataKuliahStore
{
    public function all(): array
    {
        return MataKuliah::query()->orderBy('id_mk')->get()
            ->map(fn (MataKuliah $m) => $this->baris($m))->all();
    }

    public function find(int $id): ?array
    {
        $m = MataKuliah::query()->find($id);

        return $m ? $this->baris($m) : null;
    }

    public function forKurikulum(string $kurikulumId): array
    {
        if (! ctype_digit($kurikulumId)) {
            return [];
        }

        return MataKuliah::query()->where('id_kurikulum', (int) $kurikulumId)
            ->orderBy('semester')->orderBy('kode_mk')->get()
            ->map(fn (MataKuliah $m) => $this->baris($m))->all();
    }

    public function count(string $kurikulumId): int
    {
        return ctype_digit($kurikulumId) ? MataKuliah::query()->where('id_kurikulum', (int) $kurikulumId)->count() : 0;
    }

    public function kodeExists(string $kurikulumId, string $kode, ?int $exceptId = null): bool
    {
        if (! ctype_digit($kurikulumId)) {
            return false;
        }

        return MataKuliah::query()
            ->where('id_kurikulum', (int) $kurikulumId)
            ->where('kode_mk', $kode)                       // collation utf8mb4_unicode_ci: tidak peka huruf besar/kecil
            ->when($exceptId, fn ($q) => $q->where('id_mk', '!=', $exceptId))
            ->exists();
    }

    public function create(array $data): array
    {
        $m = new MataKuliah;
        $m->forceFill($this->kolom($data))->save();

        return $this->baris($m);
    }

    public function update(int $id, array $data): ?array
    {
        $m = MataKuliah::query()->find($id);
        if (! $m) {
            return null;
        }

        $m->forceFill($this->kolom($data))->save();

        return $this->baris($m);
    }

    public function delete(int $id): bool
    {
        return MataKuliah::query()->whereKey($id)->delete() > 0;
    }

    /** Di database sudah ditangani cascadeOnDelete, tetap dibuat aman bila dipanggil langsung. */
    public function deleteByKurikulum(string $kurikulumId): void
    {
        if (ctype_digit($kurikulumId)) {
            MataKuliah::query()->where('id_kurikulum', (int) $kurikulumId)->delete();
        }
    }

    private function kolom(array $d): array
    {
        return [
            'id_kurikulum' => (int) $d['kurikulum'],
            'semester' => (int) $d['semester'],
            'kode_mk' => trim($d['kode']),
            'nama_mk' => trim($d['nama']),
            'sks_teori' => (int) $d['sks_teori'],
            'sks_praktik' => (int) $d['sks_praktek'],
            'jam_teori' => (int) $d['jam_teori'],
            'jam_praktek' => (int) $d['jam_praktek'],
        ];
    }

    private function baris(MataKuliah $m): array
    {
        return [
            'id' => (int) $m->id_mk,
            'kurikulum' => (string) $m->id_kurikulum,
            'semester' => (int) $m->semester,
            'kode' => $m->kode_mk,
            'nama' => $m->nama_mk,
            'sks_teori' => (int) $m->sks_teori,
            'sks_praktek' => (int) $m->sks_praktik,
            'jam_teori' => (int) $m->jam_teori,
            'jam_praktek' => (int) $m->jam_praktek,
            'created_at' => $m->created_at?->toDateTimeString(),
        ];
    }
}

<?php

namespace App\Repositories;

use Illuminate\Support\Facades\Storage;

/**
 * Penyimpanan mata kuliah SIMULASI berbasis file JSON (storage/app/private/admin/mata-kuliah.json).
 * Dipakai admin, kaprodi (CRUD) dan halaman publik (lewat JsonMataKuliahRepository).
 * Data awal diambil dari config/matkul.php saat file JSON belum ada.
 *
 * Satu baris: id (int), kurikulum (id kurikulum, string), semester (1-8), kode, nama,
 *             sks_teori, sks_praktek, jam_teori, jam_praktek, created_at
 *
 * TODO-DB: ganti isi method dengan Eloquent (tabel mata_kuliah). Nama method & bentuk array dijaga tetap.
 */
class MataKuliahStore
{
    public const ROMAWI = [1 => 'I', 2 => 'II', 3 => 'III', 4 => 'IV', 5 => 'V', 6 => 'VI', 7 => 'VII', 8 => 'VIII'];

    private const PATH = 'admin/mata-kuliah.json';

    /** @return array<int, array<string, mixed>> */
    public function all(): array
    {
        $disk = Storage::disk('local');

        if (! $disk->exists(self::PATH)) {
            $this->save($this->seed());
        }

        return json_decode($disk->get(self::PATH), true) ?: [];
    }

    public function find(int $id): ?array
    {
        foreach ($this->all() as $row) {
            if ((int) $row['id'] === $id) {
                return $row;
            }
        }

        return null;
    }

    /** Semua mata kuliah milik satu kurikulum. */
    public function forKurikulum(string $kurikulumId): array
    {
        return array_values(array_filter($this->all(), fn ($r) => (string) $r['kurikulum'] === $kurikulumId));
    }

    public function count(string $kurikulumId): int
    {
        return count($this->forKurikulum($kurikulumId));
    }

    /** @return int[] Semester yang sudah punya mata kuliah, urut naik. */
    public function semesters(string $kurikulumId): array
    {
        $semesters = array_unique(array_map(fn ($r) => (int) $r['semester'], $this->forKurikulum($kurikulumId)));
        sort($semesters);

        return array_values($semesters);
    }

    /** Semester pertama yang berisi data (atau 1 bila kurikulum masih kosong). */
    public function firstSemester(string $kurikulumId): int
    {
        return $this->semesters($kurikulumId)[0] ?? 1;
    }

    /** Mata kuliah pada satu semester, urut kode. */
    public function bySemester(string $kurikulumId, int $semester): array
    {
        $rows = array_values(array_filter(
            $this->forKurikulum($kurikulumId),
            fn ($r) => (int) $r['semester'] === $semester
        ));

        usort($rows, fn ($a, $b) => strnatcasecmp($a['kode'], $b['kode']));

        return $rows;
    }

    /**
     * Data untuk halaman daftar: baris yang tampil (hasil cari), jumlah semua, dan sub total
     * (sub total selalu dari SEMUA mata kuliah semester itu, bukan hasil cari).
     */
    public function listing(string $kurikulumId, int $semester, string $q = ''): array
    {
        $semua = $this->bySemester($kurikulumId, $semester);

        $tampil = $q === '' ? $semua : array_values(array_filter(
            $semua,
            fn ($r) => stripos($r['kode'], $q) !== false || stripos($r['nama'], $q) !== false
        ));

        return [
            'courses' => $tampil,
            'jumlahSemua' => count($semua),
            'romawi' => self::ROMAWI[$semester] ?? (string) $semester,
            'total' => [
                'sks_teori' => array_sum(array_column($semua, 'sks_teori')),
                'sks_praktek' => array_sum(array_column($semua, 'sks_praktek')),
                'jam_teori' => array_sum(array_column($semua, 'jam_teori')),
                'jam_praktek' => array_sum(array_column($semua, 'jam_praktek')),
            ],
        ];
    }

    /** Apakah kode sudah dipakai di kurikulum ini (abaikan baris $exceptId saat edit). */
    public function kodeExists(string $kurikulumId, string $kode, ?int $exceptId = null): bool
    {
        foreach ($this->forKurikulum($kurikulumId) as $row) {
            if (strcasecmp($row['kode'], $kode) === 0 && (int) $row['id'] !== $exceptId) {
                return true;
            }
        }

        return false;
    }

    public function create(array $data): array
    {
        $rows = $this->all();

        $row = ['id' => $this->nextId($rows)] + $this->normalize($data) + ['created_at' => now()->toDateTimeString()];

        $rows[] = $row;
        $this->save($rows);

        return $row;
    }

    public function update(int $id, array $data): ?array
    {
        $rows = $this->all();
        $found = null;

        foreach ($rows as &$row) {
            if ((int) $row['id'] === $id) {
                $row = array_merge($row, $this->normalize($data));
                $found = $row;
            }
        }
        unset($row);

        if ($found) {
            $this->save($rows);
        }

        return $found;
    }

    public function delete(int $id): bool
    {
        $rows = $this->all();
        $keep = array_values(array_filter($rows, fn ($r) => (int) $r['id'] !== $id));

        $this->save($keep);

        return count($keep) < count($rows);
    }

    /** Dipanggil saat sebuah kurikulum dihapus (di database = cascadeOnDelete). */
    public function deleteByKurikulum(string $kurikulumId): void
    {
        $this->save(array_values(array_filter(
            $this->all(),
            fn ($r) => (string) $r['kurikulum'] !== $kurikulumId
        )));
    }

    private function normalize(array $d): array
    {
        return [
            'kurikulum' => (string) $d['kurikulum'],
            'semester' => (int) $d['semester'],
            'kode' => trim($d['kode']),
            'nama' => trim($d['nama']),
            'sks_teori' => (int) $d['sks_teori'],
            'sks_praktek' => (int) $d['sks_praktek'],
            'jam_teori' => (int) $d['jam_teori'],
            'jam_praktek' => (int) $d['jam_praktek'],
        ];
    }

    private function nextId(array $rows): int
    {
        return $rows ? max(array_map('intval', array_column($rows, 'id'))) + 1 : 1;
    }

    private function save(array $rows): void
    {
        Storage::disk('local')->put(self::PATH, json_encode(array_values($rows), JSON_PRETTY_PRINT | JSON_UNESCAPED_UNICODE | JSON_UNESCAPED_SLASHES));
    }

    /** Data awal dari config/matkul.php: [id kurikulum][semester] => [kode, nama, sks_teori, sks_praktek, jam_teori, jam_praktek]. */
    private function seed(): array
    {
        $rows = [];
        $id = 1;

        foreach (config('matkul.data', []) as $kurikulumId => $semesters) {
            foreach ($semesters as $semester => $list) {
                foreach ($list as $r) {
                    $rows[] = [
                        'id' => $id++,
                        'kurikulum' => (string) $kurikulumId,   // kunci angka di PHP jadi int, paksa string
                        'semester' => (int) $semester,
                        'kode' => $r[0],
                        'nama' => $r[1],
                        'sks_teori' => $r[2],
                        'sks_praktek' => $r[3],
                        'jam_teori' => $r[4],
                        'jam_praktek' => $r[5],
                        'created_at' => now()->toDateTimeString(),
                    ];
                }
            }
        }

        return $rows;
    }
}

    /* ---------- kirim ke server ---------- */
    function request(method, url, body) {
        var opts = { method: method, headers: { 'X-CSRF-TOKEN': CSRF, 'Accept': 'application/json' } };
        if (body) opts.body = body;
        return fetch(url, opts).then(function (res) {
            return res.json().catch(function () { return {}; }).then(function (json) {
                if (!res.ok) {
                    var msg = json.errors ? Object.values(json.errors)[0][0] : (json.message || 'Terjadi kesalahan, coba lagi.');

<script>
(function () {
    var data = @json($kurikulum);
    var URL_BASE = @js(route('kaprodi.kurikulum.trk'));   // /kaprodi/kurikulum/trk
    var CSRF = document.querySelector('meta[name=csrf-token]').content;

    var $ = function (id) { return document.getElementById(id); };
    var modal = $('modal');

    // Nilai filter (cari nama + status)
    var filter = { q: '', status: 'all' };
    var editId = null, busyForm = false;

    function esc(s) {
        return String(s == null ? '' : s).replace(/[&<>"']/g, function (c) {
            return { '&': '&amp;', '<': '&lt;', '>': '&gt;', '"': '&quot;', "'": '&#39;' }[c];
        });
    }
    function icon(paths) {
        return '<svg width="16" height="16" viewBox="0 0 24 24" fill="none" stroke="currentColor" stroke-width="1.5" stroke-linecap="round" stroke-linejoin="round">' + paths + '</svg>';
    }
    var I_REFRESH = icon('<path d="M20 11a8.1 8.1 0 0 0 -15.5 -2m-.5 -4v4h4" /><path d="M4 13a8.1 8.1 0 0 0 15.5 2m.5 4v-4h-4" />');
    var I_EDIT    = icon('<path d="M7 7h-1a2 2 0 0 0 -2 2v9a2 2 0 0 0 2 2h9a2 2 0 0 0 2 -2v-1" /><path d="M20.385 6.585a2.1 2.1 0 0 0 -2.97 -2.97l-8.415 8.385v3h3l8.385 -8.415z" /><path d="M16 5l3 3" />');
    var I_TRASH   = icon('<path d="M4 7l16 0" /><path d="M10 11l0 6" /><path d="M14 11l0 6" /><path d="M5 7l1 12a2 2 0 0 0 2 2h8a2 2 0 0 0 2 -2l1 -12" /><path d="M9 7v-3a1 1 0 0 1 1 -1h6a1 1 0 0 1 1 1v3" />');
    var I_FILE    = icon('<path d="M14 3v4a1 1 0 0 0 1 1h4" /><path d="M17 21h-10a2 2 0 0 1 -2 -2v-14a2 2 0 0 1 2 -2h7l5 5v11a2 2 0 0 1 -2 2z" /><path d="M9 9l1 0" /><path d="M9 13l6 0" /><path d="M9 17l6 0" />');

    var TXT = DataTable.render.text(); // otomatis escape HTML

    function cariBaris(id) { return data.filter(function (r) { return String(r.id) === String(id); })[0]; }

    /* ---------- kirim ke server ---------- */
    function request(method, url, body) {
        var opts = { method: method, headers: { 'X-CSRF-TOKEN': CSRF, 'Accept': 'application/json' } };
        if (body) opts.body = body;
        return fetch(url, opts).then(function (res) {
            return res.json().catch(function () { return {}; }).then(function (json) {
                if (!res.ok) {
                    var msg = json.errors ? Object.values(json.errors)[0][0] : (json.message || 'Terjadi kesalahan, coba lagi.');
                    throw new Error(msg);
                }
                return json;
            });
        });
    }

    /* ---------- Filter cari + status ---------- */
    DataTable.ext.search.push(function (settings, searchData, dataIndex, r) {
        if (settings.nTable.id !== 'tabel') return true;
        return String(r.name || '').toLowerCase().indexOf(filter.q) !== -1 &&
               (filter.status === 'all' || r.status === filter.status);
    });

    var table = new DataTable('#tabel', {
        data: data,
        autoWidth: false,
        pageLength: 10,
        order: [],                       // tanpa urutan awal = urutan asli dari server
        columns: [
            {   // nama + keterangan di baris kedua (masing-masing satu baris, selebihnya "...")
                data: 'name',
                render: function (d, type, r) {
                    if (type !== 'display') return d == null ? '' : d;
                    return '<div class="truncate">' + esc(d) + '</div>' +
                           (r.description ? '<div class="truncate text-xs text-[#8A8A90]">' + esc(r.description) + '</div>' : '');
                }
            },
            { data: 'year', render: TXT, className: 'truncate' },
            {
                data: 'status',
                render: function (d, type) {
                    if (type !== 'display') return d;
                    var on = d === 'aktif';
                    return '<span class="inline-flex items-center rounded px-2 py-0.5 text-[11px] font-medium ' +
                        (on ? 'bg-[#DDF3D0] text-[#3A8A1E]' : 'bg-[#EDEDF0] text-[#55555C]') + '">' + (on ? 'Aktif' : 'Nonaktif') + '</span>';
                }
            },
            {
                data: null, orderable: false, searchable: false,
                render: function (d, type, r) {
                    var id = esc(r.id);
                    return '<div class="flex gap-2">' +
                        '<button type="button" title="Ubah status" aria-label="Ubah status" data-act="toggle" data-id="' + id + '" class="flex size-8 cursor-pointer items-center justify-center rounded-md bg-[#FFF1D6] text-[#C77700] hover:bg-[#FCE7BD]">' + I_REFRESH + '</button>' +
                        '<button type="button" title="Edit" aria-label="Edit" data-act="edit" data-id="' + id + '" class="flex size-8 cursor-pointer items-center justify-center rounded-md bg-[#D4E3FF] text-[#1F73FF] hover:bg-[#C2D8FF]">' + I_EDIT + '</button>' +
                        '<button type="button" title="Hapus" aria-label="Hapus" data-act="delete" data-id="' + id + '" class="flex size-8 cursor-pointer items-center justify-center rounded-md bg-[#FFD9D9] text-[#E5484D] hover:bg-[#FFC9C9]">' + I_TRASH + '</button>' +
                    '</div>';
                }
            },
            {
                data: null, orderable: false, searchable: false,
                render: function (d, type, r) {
                    var ada = !!r.pdf_url;
                    return '<button type="button" title="' + (ada ? 'Buka dokumen PDF' : 'Belum ada dokumen. Klik Edit untuk mengunggahnya.') + '" data-act="pdf" data-id="' + esc(r.id) + '"' + (ada ? '' : ' aria-disabled="true"') +
                        ' class="flex h-7 items-center gap-1 rounded-md border border-[#1F73FF] bg-white px-2.5 text-xs font-medium text-[#1F73FF]' + (ada ? ' cursor-pointer hover:bg-[#F2F7FF]' : ' cursor-not-allowed opacity-40') + '">' + I_FILE + 'Lihat PDF</button>';
                }
            }
        ],
        // Tooltip: arahkan kursor ke teks yang terpotong "..." -> muncul teks lengkapnya
        createdRow: function (tr, r) {
            var td = tr.children;
            td[0].title = (r.name || '') + (r.description ? '\n' + r.description : '');
            td[1].title = r.year || '';
        },
        // Hanya paginasi (tengah bawah). Kotak cari, "show entries", dan info bawaan dimatikan
        layout: { topStart: null, topEnd: null, bottomStart: null, bottomEnd: null, bottom: 'paging' },
        pagingType: 'simple_numbers',
        language: {
            emptyTable: 'Tidak ada kurikulum.',
            zeroRecords: 'Tidak ada kurikulum.',
            paginate: { previous: 'Sebelumnya', next: 'Berikutnya' }
        }
    });

    // Sembunyikan paginasi kalau hanya ada 1 halaman
    function togglePager() {
        var p = document.querySelector('#tabel-wrap .dt-paging');
        if (p) p.style.display = table.page.info().pages > 1 ? '' : 'none';
    }
    table.on('draw.dt', togglePager);
    togglePager();

    // Muat ulang isi tabel dari array `data`. tetapDiHalaman=true -> tetap di halaman sekarang; false -> kembali ke halaman 1
    function muatUlang(tetapDiHalaman) {
        table.clear().rows.add(data).draw(!tetapDiHalaman);
    }

    $('statusFilter').addEventListener('change', function (e) { filter.status = e.target.value; table.draw(); });

    /* ---------- Pop-up tambah / edit ---------- */
    function showError(msg) { var e = $('fError'); e.textContent = msg; e.classList.remove('hidden'); }

    function openForm(row) {

        $('fName').focus();

    function closeForm() { modal.classList.add('hidden'); modal.classList.remove('flex'); }
    function formTerbuka() { return !modal.classList.contains('hidden'); }

    function save() {
        if (busyForm) return;
        var name = $('fName').value.trim(), desc = $('fDesc').value.trim(), year = $('fYear').value.trim();
        if (!name || !year) { showError('Nama kurikulum dan tahun terbit wajib diisi.'); return; }
        if (!/^\d{4}$/.test(year)) { showError('Tahun terbit harus 4 angka, mis. 2026.'); return; }

        var fd = new FormData();
        fd.append('name', name); fd.append('description', desc); fd.append('year', year);
        var file = $('fPdf').files[0];
        if (file) fd.append('pdf', file);
        if (editId) fd.append('_method', 'PUT');

        var idEdit = editId;
        busyForm = true; $('btnSave').disabled = true;
        request('POST', idEdit ? URL_BASE + '/' + encodeURIComponent(idEdit) : URL_BASE, fd)
            .then(function (json) {
                if (idEdit) {
                    data = data.map(function (r) { return String(r.id) === String(idEdit) ? json.data : r; });
                } else {
                    data.unshift(json.data);
                }
                closeForm();
                muatUlang(!!idEdit);          // edit: tetap di halaman; tambah: ke halaman 1 (data baru ada di paling atas)
            })
            .catch(function (err) { showError(err.message); })
            .then(function () { busyForm = false; $('btnSave').disabled = false; });
    }

    $('btnAdd').addEventListener('click', function () { openForm(null); });
    $('btnCancel').addEventListener('click', closeForm);
    $('btnSave').addEventListener('click', save);
    modal.addEventListener('click', function (e) { if (e.target === modal) closeForm(); });
    document.addEventListener('keydown', function (e) {
        if (!formTerbuka()) return;
        if (e.key === 'Escape') closeForm();
        if (e.key === 'Enter' && e.target.tagName === 'INPUT' && e.target.type !== 'file') save();
    });

    /* ---------- Hapus: modal konfirmasi, lalu DELETE lewat fetch; tabel dimuat ulang & kembali ke halaman 1 ---------- */
    var hd = $('modalHapus'), btnYa = $('modalHapusYa'), pesanError = $('modalHapusError');
    var idHapus = null, busyHapus = false;

    function hapusBaris(id) {
        return request('DELETE', URL_BASE + '/' + encodeURIComponent(id)).then(function () {
            data = data.filter(function (r) { return String(r.id) !== String(id); });
            muatUlang(false);                 // kembali ke halaman 1
        });
    }

    function bukaHapus(row) {
        idHapus = row.id;
        if (typeof hd.showModal !== 'function') {                      // cadangan untuk browser lama
            if (confirm('Hapus kurikulum "' + row.name + '"? Semua mata kuliah di kurikulum ini juga ikut terhapus.')) {
                hapusBaris(idHapus).catch(function (err) { alert(err.message); });
            }
            return;
        }
        $('modalHapusNama').textContent = '"' + row.name + '"';
        pesanError.hidden = true;
        hd.showModal();                                                // fokus awal di tombol Batal; Esc menutup modal
    }

    $('modalHapusBatal').addEventListener('click', function () { hd.close(); });
    hd.addEventListener('click', function (e) { if (e.target === hd) hd.close(); });   // klik area gelap = batal
    hd.addEventListener('close', function () { idHapus = null; busyHapus = false; btnYa.disabled = false; btnYa.textContent = 'Hapus'; });
    btnYa.addEventListener('click', function () {
        if (idHapus == null || busyHapus) return;
        busyHapus = true; btnYa.disabled = true; btnYa.textContent = 'Menghapus...';   // cegah klik ganda
        hapusBaris(idHapus)
            .then(function () { hd.close(); })
            .catch(function (err) {
                pesanError.textContent = err.message; pesanError.hidden = false;
                busyHapus = false; btnYa.disabled = false; btnYa.textContent = 'Hapus';
            });
    });

    /* ---------- Aksi di tabel ---------- */
    $('tabel').addEventListener('click', function (e) {
        var b = e.target.closest('button[data-act]');
        if (!b || b.getAttribute('aria-disabled') === 'true') return;
        var row = cariBaris(b.dataset.id); if (!row) return;

        if (b.dataset.act === 'toggle') {
            request('PATCH', URL_BASE + '/' + encodeURIComponent(row.id) + '/status').then(function (json) {
                data = data.map(function (r) { return String(r.id) === String(row.id) ? json.data : r; });
                muatUlang(true);              // tetap di halaman sekarang
            }).catch(function (err) { alert(err.message); });
        } else if (b.dataset.act === 'edit') {
            openForm(row);
        } else if (b.dataset.act === 'delete') {
            bukaHapus(row);
        } else if (b.dataset.act === 'pdf' && row.pdf_url) {
            window.open(row.pdf_url, '_blank', 'noopener');
        }
    });
})();
